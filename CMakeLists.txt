cmake_minimum_required(VERSION 3.24)
project(AsterBook VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASTERBOOK_ASAN_UBSAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ASTERBOOK_TSAN "Enable ThreadSanitizer (separate build)" OFF)
option(ASTERBOOK_FUZZ "Build libFuzzer target (Clang only)" OFF)
set(ASTERBOOK_FUZZ_RUNTIME "" CACHE FILEPATH "Optional standalone libFuzzer archive")
if(ASTERBOOK_ASAN_UBSAN AND ASTERBOOK_TSAN)
  message(FATAL_ERROR "ASan and TSan must be built separately")
endif()
if(ASTERBOOK_ASAN_UBSAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(ASTERBOOK_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(asterbook_core src/book.cpp src/engine.cpp src/journal.cpp)
target_include_directories(asterbook_core PUBLIC include)
target_compile_options(asterbook_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(asterbook_gateway src/protocol.cpp src/gateway.cpp src/tls.cpp)
target_include_directories(asterbook_gateway PUBLIC include)
target_link_libraries(asterbook_gateway PUBLIC asterbook_core Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
target_compile_options(asterbook_gateway PRIVATE -Wall -Wextra -Wpedantic)
add_executable(asterbook src/main.cpp)
target_link_libraries(asterbook PRIVATE asterbook_gateway)
add_library(asterbook_itch_lib src/itch.cpp)
target_include_directories(asterbook_itch_lib PUBLIC include)
add_executable(asterbook_itch tools/itch_main.cpp)
target_link_libraries(asterbook_itch PRIVATE asterbook_itch_lib)
add_executable(recovery_verify tools/recovery_verify.cpp)
target_link_libraries(recovery_verify PRIVATE asterbook_core Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  foreach(name core recovery protocol queue)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE asterbook_gateway Threads::Threads)
    target_compile_options(${name}_test PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name}_test)
  endforeach()
  add_executable(recovery_fault_test tests/recovery_fault_test.cpp src/book.cpp src/engine.cpp src/journal.cpp)
  target_include_directories(recovery_fault_test PRIVATE include)
  target_compile_definitions(recovery_fault_test PRIVATE ASTERBOOK_TEST_FAULTS)
  target_compile_options(recovery_fault_test PRIVATE -UNDEBUG)
  target_link_libraries(recovery_fault_test PRIVATE Threads::Threads)
  add_test(NAME recovery_fault COMMAND recovery_fault_test)
  add_test(NAME tls COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/tls_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME network COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/network_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME network_pressure COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/network_pressure_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME network_stress COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/network_stress_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME network_shards COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/network_shards_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME differential COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/reference/differential.py --server $<TARGET_FILE:asterbook>)
  set_tests_properties(network network_shards network_pressure network_stress differential PROPERTIES TIMEOUT 120)
  add_executable(parser_smoke fuzz/parser_smoke.cpp)
  target_link_libraries(parser_smoke PRIVATE asterbook_gateway)
  add_test(NAME parser_smoke COMMAND parser_smoke)
  add_executable(itch_sequence_test tests/itch_sequence_test.cpp)
  target_link_libraries(itch_sequence_test PRIVATE asterbook_itch_lib)
  target_compile_options(itch_sequence_test PRIVATE -UNDEBUG)
  add_test(NAME itch_sequence COMMAND itch_sequence_test)
  add_test(NAME itch_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/itch_native_test.py $<TARGET_FILE:asterbook_itch>)
  add_test(NAME itch COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/reference/test_itch.py)
  add_test(NAME recovery_backup COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/recovery_backup_test.py $<TARGET_FILE:asterbook> $<TARGET_FILE:recovery_verify>)
  add_test(NAME tls_rotation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/tls_rotation_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME monitor_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/monitor_integration_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME runtime_ops COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/runtime_ops_test.py $<TARGET_FILE:asterbook>)
  add_test(NAME backup_service COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/backup_service_test.py)
  add_test(NAME release_tooling COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/release_test.py)
  set_tests_properties(tls tls_rotation runtime_ops monitor_integration recovery_backup backup_service release_tooling PROPERTIES TIMEOUT 120)
  add_test(NAME benchmark_harness COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/benchmark_harness_test.py)
endif()
if(ASTERBOOK_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "libFuzzer target requires Clang")
  endif()
  if(ASTERBOOK_TSAN)
    message(FATAL_ERROR "Fuzzing and TSan require separate builds")
  endif()
  if(ASTERBOOK_FUZZ_RUNTIME)
    set(FUZZ_COMPILE_FLAGS -fsanitize=fuzzer-no-link,address,undefined)
    set(FUZZ_LINK_FLAGS -fsanitize=address,undefined)
    set(FUZZ_LIBRARY "${ASTERBOOK_FUZZ_RUNTIME}")
  else()
    set(FUZZ_COMPILE_FLAGS -fsanitize=fuzzer,address,undefined)
    set(FUZZ_LINK_FLAGS -fsanitize=fuzzer,address,undefined)
  endif()
  add_executable(itch_fuzz fuzz/itch_fuzz.cpp src/itch.cpp)
  target_include_directories(itch_fuzz PRIVATE include)
  target_compile_options(itch_fuzz PRIVATE ${FUZZ_COMPILE_FLAGS} -fno-omit-frame-pointer)
  target_link_options(itch_fuzz PRIVATE ${FUZZ_LINK_FLAGS})
  target_link_libraries(itch_fuzz PRIVATE ${FUZZ_LIBRARY})
  add_executable(parser_fuzz fuzz/protocol_fuzz.cpp src/protocol.cpp)
  target_include_directories(parser_fuzz PRIVATE include)
  target_compile_options(parser_fuzz PRIVATE ${FUZZ_COMPILE_FLAGS} -fno-omit-frame-pointer)
  target_link_options(parser_fuzz PRIVATE ${FUZZ_LINK_FLAGS})
  target_link_libraries(parser_fuzz PRIVATE ${FUZZ_LIBRARY})
endif()
add_executable(core_bench bench/core_bench.cpp)
target_link_libraries(core_bench PRIVATE asterbook_core)
option(ASTERBOOK_GOOGLE_BENCHMARK "Build optional Google Benchmark microbenchmarks (downloads dependency)" OFF)
if(ASTERBOOK_GOOGLE_BENCHMARK)
  include(FetchContent)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(google_benchmark
    URL https://github.com/google/benchmark/archive/refs/tags/v1.9.4.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    URL_HASH SHA256=b334658edd35efcf06a99d9be21e4e93e092bd5f95074c1673d5c8705d95c104)
  FetchContent_MakeAvailable(google_benchmark)
  add_executable(microbench bench/microbench.cpp)
  target_link_libraries(microbench PRIVATE asterbook_core benchmark::benchmark)
endif()
add_executable(durable_bench bench/durable_bench.cpp)
target_link_libraries(durable_bench PRIVATE asterbook_core)
